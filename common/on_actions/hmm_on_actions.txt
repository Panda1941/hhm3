on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = {
					is_playable_country = yes
				}
				add_ideas = {
					peace_training
				}
			}
			every_country = {
				every_army_leader = {
					limit = {
						NOT = { 
							has_trait = no_grind
						}
					}
					#add_unit_leader_trait = no_grind
				}
			}
			every_country = {
				limit = {
					OR = {
						ai_annex = yes
						ai_rk = yes
						ai_manpower = yes
						ai_black = yes
						tag = SWE
					}
				}
				country_lock_all_division_template = yes
				add_ideas = {
					ai_tradelaw
				}
			}
			SOV = { 
				country_event = {
					id = mexico.21 # Mexico returns trotsky
					days = 396 # Feb 37
				}
			}
			every_country = {
				limit = {
					OR = {
						ai_annex = yes
						ai_black = yes
					}
				}
				add_ideas = {
					ai_annex
				}
			}
			every_country = {
				limit = {
					OR = {
						ai_manpower = yes
					}
				}
				add_ideas = {
					ai_war
				}
			}
			every_country = {
				add_ideas = {
					Spies_no_capture
				}
			}
			GER = {
				country_event = {
					id = DOD_yugoslavia.20
					days = 912
				}
			}
			FRA = {
				country_event = {
					id = DOD_yugoslavia.20
					days = 912
				}
			}
			##### Turning King Edward's abdication into an on_startup event instead of MTTH to help performance####
			ENG = {
				country_event = {
					id = britain.16
					days = 20
				}
				country_event = {
					id = mtg_britain.1
					days = 190
				}
				country_event = {
					id = britain.22
					days = 513
				}
				country_event = {
					id = DOD_yugoslavia.20
					days = 912
				}
			}
			ITA = {
				country_event = {
					id = DOD_yugoslavia.20
					days = 912
				}
			}
		}
	}
	on_monthly = {
		effect = {
			every_country = {
				limit = {
					is_playable_country = yes
					NOT = {
						OR = {
							has_war = yes
							has_idea = peace_training
						}
					}
				}
				add_ideas = peace_training
			}
			if = {
				limit = {
					HOL = {
						has_equipment = {
							convoy_1 < 1000
						}
						has_capitulated = yes
					}
				}
				HOL = {
					add_equipment_to_stockpile = {
						type = convoy
						amount = 1000
					}
				}
			}
		}
	}
	on_capitulation = {
		effect = {
			hmm_capitulations_on_actions = yes
		}
	}
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					FROM = { tag = FRA }
					FROM.FROM = { is_core_of = FRA }
					OR ={  AND ={  FRA = { surrender_progress > 0.5 NOT = { controls_state = 16 } } }	FRA = { surrender_progress	> 0.7	}	 }
					NOT = { FRA = { has_country_flag = france_tp} }
				}


				FRA = { 
					set_country_flag = france_tp
					transfer_units_fraction = {
						target = COG
						size = 1
						stockpile_ratio = 1
						army_ratio = 1
						navy_ratio = 1
						air_ratio = 1
						keep_unit_leaders_trigger = {
							always = yes
						}
					}					
				}
			}
		}
	}
	on_war_relation_added = {
        effect = {
            if = {
                limit = {
                    tag = JAP
                    FROM = {
                        is_in_faction_with = ENG
                    }
                }
                every_country = {
                    limit = {
                        is_in_faction_with = ENG
                    }
                    add_to_war = {
                        targeted_alliance = FROM
                        enemy = JAP
                        hostility_reason = asked_to_join
                    }
                }
                every_country = {
                    limit = {
                        is_in_faction_with = JAP
                    }
                    add_to_war = {
                        targeted_alliance = ROOT
                        enemy = FROM
                        hostility_reason = asked_to_join
                    }
                }
            }
			#VP System
			if = {
				limit = {
					FROM = {
						original_tag = SOV
					}
				}
				initilize_victory_system = yes
			}
			if = {
                limit = {
                    tag = GER
                    FROM = {
                       tag = SOV
                    }
                }
                every_country = {
                    limit = {
                        is_in_faction_with = GER
                    }
                    add_to_war = {
                        targeted_alliance = ROOT
                        enemy = SOV
                        hostility_reason = asked_to_join
                    }
                }
            }
        }
    }
	on_declare_war = {
		effect = {
			if = {
				limit = {
					FROM = {
						tag = POL
					}
				}
				ENG = {
					add_to_faction = BRA
				}
				HOL ={
					diplomatic_relation = {
						country = GER
						relation = embargo
						active = yes
					}
				}
			}

			if = {
				limit = {
					FROM = {
						allies = yes
						NOT = { tag = SOV}
					}
				}
				ENG = {
					add_to_faction = FROM
				}
			}

            if = {
                limit = {
                    FROM = {
                        tag = BEL
                    }
                }
                BEL = {
                    load_oob = "BEL_CHROM_SPAWN"
                }
            }

            if = {
                limit = {
                    FROM = {
                        tag = DEN
                    }
                }
                DEN = {
                    load_oob = "DEN_CHROM_SPAWN"
                }
            }

            if = {
                limit = {
                    FROM = {
                        tag = HOL
                    }
                }
                HOL = {
                    load_oob = "HOL_CHROM_SPAWN"
                }
            }

            if = {
                limit = {
                    FROM = {
                        tag = LUX
                    }
                }
                LUX = {
                    load_oob = "LUX_CHROM_SPAWN"
                }
            }

            if = {
                limit = {
                    FROM = {
                        tag = NOR
                    }
                }
                NOR = {
                    load_oob = "NOR_CHROM_SPAWN"
                }
            }

			if = {
                limit = {
                    FROM = {
                        tag = POL
                    }
                }
                POL = {
                   load_oob = "POL_CHROM_SPAWN"
                }
            }
		}
	}
	on_startup = {
		effect = {
			if = {
				limit ={
					has_game_rule = {
						rule = playable_iraq
						option = no
					}
				}
				ENG = { 
					annex_country = { target = IRQ transfer_troops = yes }
					every_state = {
						limit = {
							is_core_of = IRQ
						}
						set_compliance = 80
					}
				}
			}

			## ANNEX STUFF LIKE MEXICO CUZ ITS NOT PLAYABLE ATM
			WST = {
				annex_country = { target = MEX transfer_troops = no }
			}

			# OTHER SETUP STUFF
			every_country = {	# Adding starting XP
				limit = {
					is_playable_country = yes
				}
				army_experience = 10
				navy_experience = 10
				air_experience = 10
			}

			# SPAIN DMZ
			every_state = {
				limit = {
					is_owned_and_controlled_by = SPR
				}
				add_building_construction = { type = anti_air_building level = 2 instant_build = yes }
				add_dynamic_modifier = { modifier = SPR_cant_build_airports }
			}

			# NO TRADEBACKS
			every_country = {
				limit = {
					is_playable_country = yes
				}
				add_ideas = no_tradebacks
			}
		}
	}
